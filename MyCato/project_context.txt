import React, { useState } from "react";
import {
	Monitor,
	Sun,
	Moon,
	Globe,
	User,
	Zap,
	Plus,
	FileJson, // New icon for the button
	Copy,
	Check,
	X,
} from "lucide-react";
import "./assets/App.css";

// Context
import { ConfigProvider, useConfig } from "./context/ConfigContext";

// Tabs
import GlobalTab from "./components/tabs/GlobalTab";
import ActionManagerTab from "./components/tabs/ActionManagerTab";
import ProfileContainer from "./components/profiles/ProfileContainer";

// --- INTERNAL JSON MODAL COMPONENT ---
const JsonPreviewModal = ({ isOpen, onClose, data }) => {
	const [copied, setCopied] = useState(false);

	if (!isOpen) return null;

	const copyJson = () => {
		navigator.clipboard.writeText(JSON.stringify(data, null, 2));
		setCopied(true);
		setTimeout(() => setCopied(false), 2000);
	};

	return (
		<div className="modal-overlay">
			<div className="modal-box-large" style={{ height: "80vh" }}>
				<div className="modal-header-compact" style={{ justifyContent: "space-between" }}>
					<h3 style={{ margin: 0, display: "flex", alignItems: "center", gap: "0.5rem" }}>
						<FileJson size={20} className="text-indigo-500" />
						Config JSON Preview
					</h3>
					<div style={{ display: "flex", gap: "0.5rem" }}>
						<button className="btn" onClick={copyJson}>
							{copied ? <Check size={16} /> : <Copy size={16} />}
							{copied ? "Copied" : "Copy to Clipboard"}
						</button>
						<button className="btn-ghost" onClick={onClose}>
							<X size={20} />
						</button>
					</div>
				</div>
				<div className="modal-body" style={{ padding: 0, overflow: "hidden", display: "flex" }}>
					<pre className="code-block-full">
						{JSON.stringify(data, null, 2)}
					</pre>
				</div>
			</div>
		</div>
	);
};

const ConfigEditorShell = () => {
	const [activeTab, setActiveTab] = useState("global");
	const [showJson, setShowJson] = useState(false); // Modal State

	const {
		data,
		isAdvanced,
		setIsAdvanced,
		isDarkMode,
		setIsDarkMode,
		resetConfig,
		addConfigItem,
	} = useConfig();

	return (
		<div className="app-wrapper">
			<header className="header-bar">
				<div className="header-title">
					<Monitor className="text-indigo-500" /> Config Editor
				</div>
				<div className="header-actions">
					<button
						className="btn btn-ghost"
						onClick={() => setIsDarkMode(!isDarkMode)}
						title="Toggle Dark Mode"
					>
						{isDarkMode ? <Sun size={20} /> : <Moon size={20} />}
					</button>
					
					{/* NEW: JSON Preview Button */}
					<button 
						className="btn btn-ghost"
						onClick={() => setShowJson(true)}
						title="View JSON"
					>
						<FileJson size={20} />
					</button>

					<button className="btn" onClick={resetConfig}>
						Reset
					</button>
					<button
						className="btn btn-primary"
						onClick={() => console.log(data)}
					>
						Save Config
					</button>
				</div>
			</header>

			<div className="sticky-tabs-bar">
				<div className="tabs-container">
					<button
						className={`tab-btn ${activeTab === "global" ? "active" : ""}`}
						onClick={() => setActiveTab("global")}
					>
						<Globe size={18} /> Global
					</button>
					<button
						className={`tab-btn ${activeTab === "profiles" ? "active" : ""}`}
						onClick={() => setActiveTab("profiles")}
					>
						<User size={18} /> Profiles
					</button>
					<button
						className={`tab-btn ${activeTab === "actions" ? "active" : ""}`}
						onClick={() => setActiveTab("actions")}
					>
						<Zap size={18} /> Actions
					</button>
				</div>

				<div style={{ display: "flex", gap: "1rem", alignItems: "center" }}>
					{activeTab === "profiles" && (
						<button
							className="btn btn-primary"
							style={{ padding: "0.4rem 0.8rem" }}
							onClick={() => addConfigItem(["profiles"])}
						>
							<Plus size={16} /> New Profile
						</button>
					)}
					{activeTab !== "actions" && (
						<label className="toggle-switch">
							<span
								style={{
									fontSize: "0.85rem",
									marginRight: "8px",
									color: "var(--text-sub)",
								}}
							>
								Advanced Mode
							</span>
							<input
								className="toggle-input"
								type="checkbox"
								checked={isAdvanced}
								onChange={(e) => setIsAdvanced(e.target.checked)}
							/>
							<div className="toggle-track">
								<div className="toggle-thumb" />
							</div>
						</label>
					)}
				</div>
			</div>

			<div className="main-content">
				{/* SINGLE PANE LAYOUT */}
				<div className="content-container">
					<div className="main-card">
						<div className="scroll-area">
							{activeTab === "global" && <GlobalTab />}

							{activeTab === "profiles" && (
								<div style={{ display: "flex", flexDirection: "column", gap: "1rem" }}>
									<div
										style={{
											display: "flex",
											justifyContent: "space-between",
											alignItems: "center",
											paddingBottom: "1rem",
											borderBottom: "1px solid var(--border)",
										}}
									>
										<h3 style={{ margin: 0, fontSize: "1.1rem" }}>
											Device Profiles ({data.profiles.length})
										</h3>
										<button
											className="btn btn-primary"
											onClick={() => addConfigItem(["profiles"])}
										>
											<Plus size={16} /> Add Profile
										</button>
									</div>

									<div>
										{data.profiles.map((profile, index) => (
											<ProfileContainer
												key={index}
												index={index}
												data={profile}
											/>
										))}
									</div>

									{data.profiles.length === 0 && (
										<div
											style={{
												textAlign: "center",
												padding: "2rem",
												color: "var(--text-sub)",
											}}
										>
											No profiles found. Click "Add Profile" to create one.
										</div>
									)}
								</div>
							)}

							{activeTab === "actions" && <ActionManagerTab />}
						</div>
					</div>
				</div>
			</div>

			{/* JSON MODAL */}
			<JsonPreviewModal 
				isOpen={showJson} 
				onClose={() => setShowJson(false)} 
				data={data} 
			/>
		</div>
	);
};

const App = () => (
	<ConfigProvider>
		<ConfigEditorShell />
	</ConfigProvider>
);

export default App;

// src/components/modals/ActionModal.jsx
import { useState, useEffect } from "react";
import { ACTION_SCHEMA } from "../../constants/schemas";
import { SCROLL_TYPES, AXIS_OPTS } from "../../constants/mappings";
import {
	X,
	Save,
	Command,
	MousePointer,
	Keyboard as KeyboardIcon,
} from "lucide-react";
import { getHidLabel } from "../../utils/helpers";
import { ModifierPicker } from "../ModifierPicker";
import { MousePickerModal } from "./MousePickerModal";
import { KeyPickerModal } from "./KeyPickerModal";

export const ActionModal = ({ isOpen, onClose, onSave, initialData }) => {
	const [label, setLabel] = useState("");
	const [command, setCommand] = useState("noop");
	const [args, setArgs] = useState([]);
	const [isKeyPickerOpen, setKeyPickerOpen] = useState(false);
	const [isMousePickerOpen, setMousePickerOpen] = useState(false);

	useEffect(() => {
		if (isOpen) {
			if (initialData) {
				setLabel(initialData.label);
				setCommand(initialData.command);
				setArgs(initialData.args || []);
			} else {
				setLabel("");
				setCommand("noop");
				setArgs([]);
			}
		}
	}, [isOpen, initialData]);

	const handleSave = (e) => {
		e.preventDefault();
		const finalLabel = label || ACTION_SCHEMA[command].label;
		onSave({ label: finalLabel, command, args });
		onClose();
	};

	const setArgVal = (idx, val) => {
		const newArgs = [...args];
		newArgs[idx] = val;
		setArgs(newArgs);
	};

	const renderButtonConfig = () => {
		const actor = args[0] ?? 0;
		const action = args[1] ?? "tap";
		const nested = Array.isArray(args[2]) ? args[2] : [args[2] || 0, 0];
		const btnId = nested[0];
		const mask = nested[1];
		const updateNested = (idx, val) => {
			const newNested = [...nested];
			newNested[idx] = val;
			const newArgs = [...args];
			newArgs[0] = actor;
			newArgs[1] = action;
			newArgs[2] = newNested;
			setArgs(newArgs);
		};

		return (
			<div
				style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: "2rem" }}
			>
				{/* LEFT COLUMN: Inputs & Button Selection */}
				<div style={{ display: "flex", flexDirection: "column", gap: "1rem" }}>
					<div
						style={{
							display: "grid",
							gridTemplateColumns: "1fr 1fr",
							gap: "1rem",
						}}
					>
						<div className="form-group">
							<label>Actor</label>
							<select
								className="input-base"
								value={actor}
								onChange={(e) => {
									const newActor = Number(e.target.value);
									const newArgs = [newActor, action, [0, 0]];
									setArgs(newArgs);
								}}
							>
								<option value={0}>Mouse</option>
								<option value={1}>Keyboard</option>
							</select>
						</div>
						<div className="form-group">
							<label>Action</label>
							<select
								className="input-base"
								value={action}
								onChange={(e) => setArgVal(1, e.target.value)}
							>
								{[
									"tap",
									"double_tap",
									"press",
									"release",
									"toggle",
									"hold_until_idle",
								].map((o) => (
									<option
										key={o}
										value={o}
									>
										{o}
									</option>
								))}
							</select>
						</div>
					</div>

					<div
						className="form-group"
						style={{ flex: 1, display: "flex", flexDirection: "column" }}
					>
						<label>{actor === 0 ? "Mouse Button" : "Keyboard Key"}</label>
						<button
							type="button"
							className="key-preview-btn"
							onClick={() =>
								actor === 0 ? setMousePickerOpen(true) : setKeyPickerOpen(true)
							}
						>
							{actor === 0 ? (
								<MousePointer
									size={24}
									className="text-primary"
								/>
							) : (
								<KeyboardIcon
									size={24}
									className="text-primary"
								/>
							)}
							<div style={{ textAlign: "left" }}>
								<div className="key-sub-text">Selected Input</div>
								<div className="key-big-text">{getHidLabel(btnId)}</div>
							</div>
						</button>
					</div>
				</div>

				{/* RIGHT COLUMN: Modifiers */}
				<div className="form-group">
					<label>Modifiers (Bitmask: {mask})</label>
					<div
						style={{
							background: "var(--bg-input)",
							padding: "1rem",
							borderRadius: "8px",
							border: "1px solid var(--input-border)",
						}}
					>
						<ModifierPicker
							value={mask}
							onChange={(val) => updateNested(1, val)}
						/>
					</div>
				</div>

				<MousePickerModal
					isOpen={isMousePickerOpen}
					selectedValue={btnId}
					onClose={() => setMousePickerOpen(false)}
					onSelect={(val) => {
						updateNested(0, val);
						setTimeout(() => setMousePickerOpen(false), 200);
					}}
				/>
				<KeyPickerModal
					isOpen={isKeyPickerOpen}
					selectedValue={btnId}
					onClose={() => setKeyPickerOpen(false)}
					onSelect={(val) => {
						updateNested(0, val);
						setTimeout(() => setKeyPickerOpen(false), 200);
					}}
				/>
			</div>
		);
	};

	const renderScrollConfig = () => {
		const type = args[0] || "fixed";
		const handleTypeChange = (newType) => {
			if (newType === "fixed") setArgs(["fixed", 3]);
			else if (newType === "classic")
				setArgs(["classic", ["pitch", 5], ["yaw", 20]]);
			else if (newType === "toggle") setArgs(["toggle", ["pitch", 5]]);
			else if (newType === "until_idle") setArgs(["until_idle", ["pitch", 5]]);
		};
		return (
			<div className="schema-fields">
				<div
					style={{
						display: "grid",
						gridTemplateColumns: "1fr 1fr",
						gap: "1rem",
						marginBottom: "1rem",
					}}
				>
					<div className="form-group">
						<label>Scroll Type</label>
						<select
							className="input-base"
							value={type}
							onChange={(e) => handleTypeChange(e.target.value)}
						>
							{SCROLL_TYPES.map((t) => (
								<option
									key={t}
									value={t}
								>
									{t}
								</option>
							))}
						</select>
					</div>
					{type === "fixed" && (
						<div className="form-group">
							<label>Lines (Neg for up)</label>
							<input
								type="number"
								className="input-base"
								value={args[1] || 0}

const ProfileContainer = ({ index, data }) => {
	const { setConfigValue, deleteConfigItem } = useConfig();

	const [isCollapsed, setIsCollapsed] = useState(false);
	const [isEditing, setIsEditing] = useState(false);
	const [tempName, setTempName] = useState(data.name);

	useEffect(() => {
		setTempName(data.name);
	}, [data.name]);

	const handleSaveName = (e) => {
		e.stopPropagation();
		setConfigValue(["profiles", index, "name"], tempName);
		setIsEditing(false);
	};

	const handleCancelEdit = (e) => {
		e.stopPropagation();
		setTempName(data.name);
		setIsEditing(false);
	};

	const handleDelete = (e) => {
		e.stopPropagation();
		if (window.confirm(`Are you sure you want to delete "${data.name}"?`)) {
			deleteConfigItem(["profiles", index]);
		}
	};

	return (
		<div style={{ marginBottom: "1rem" }}>
			{/* Header */}
			<div
				className="group-header"
				onClick={() => !isEditing && setIsCollapsed(!isCollapsed)}

								onChange={(e) => setTempName(e.target.value)}

import React, { useState } from "react";
import { Plus, Terminal, Pencil, Trash2 } from "lucide-react";
import { useConfig } from "../../context/ConfigContext";
import { ACTION_SCHEMA } from "../../constants/schemas";
import { getActionSummary } from "../../utils/helpers";
import { ActionModal } from "../modals/ActionModal";
import "./ActionManager.css";
const ActionManagerTab = () => {
	const { actionList, updateActions } = useConfig();
	const [modalOpen, setModalOpen] = useState(false);
	const [editingIndex, setEditingIndex] = useState(null);

	const handleSave = (newAction) => {
		const target =
			editingIndex !== null ? [...actionList] : [...actionList, newAction];
		if (editingIndex !== null) target[editingIndex] = newAction;
		updateActions(target);
	};

	const handleDelete = (idx) => {
		const target = actionList.filter((_, i) => i !== idx);
		updateActions(target);
	};

	return (
		<div style={{ display: "flex", flexDirection: "column", gap: "1rem" }}>
			<div
				style={{
					display: "flex",
					justifyContent: "space-between",
					alignItems: "center",
				}}
			>
				<h3 style={{ margin: 0, fontSize: "1.1rem" }}>
					Defined Macros ({actionList.length})
				</h3>
				<button
					className="btn btn-primary"
					onClick={() => {
						setEditingIndex(null);
						setModalOpen(true);
					}}
				>
					<Plus size={16} /> New Action
				</button>
			</div>
			<ul className="action-list-ul">
				{actionList.map((act, i) => (
					<li
						key={i}
						className="action-item"
					>
						<div className="item-left">
							<div
								style={{
									fontFamily: "monospace",
									color: "var(--text-sub)",
									fontSize: "0.8rem",
									width: "20px",
									flexShrink: 0,
								}}
							>
								{i}
							</div>
							<div className="icon-box">
								{ACTION_SCHEMA[act.command]?.icon || <Terminal size={20} />}
							</div>
							<div className="action-content">
								<h3>{act.label}</h3>
								<div className="meta-row">
									<span className="command-pill">{act.command}</span>
									{getActionSummary(act)}
								</div>
							</div>
						</div>
						<div style={{ display: "flex", gap: "5px", marginLeft: "1rem" }}>
							<button
								className="btn btn-ghost"
								onClick={() => {
									setEditingIndex(i);
									setModalOpen(true);
								}}
							>
								<Pencil size={18} />
							</button>
							<button
								className="btn btn-ghost btn-danger"
								onClick={() => handleDelete(i)}
							>
								<Trash2
									size={18}
									color="var(--danger)"
								/>
							</button>
						</div>
					</li>
				))}
			</ul>
			<ActionModal
				isOpen={modalOpen}
				initialData={editingIndex !== null ? actionList[editingIndex] : null}
				onClose={() => setModalOpen(false)}
				onSave={handleSave}
			/>
		</div>
	);
};

export default ActionManagerTab;

import React, { useState, useEffect } from "react";
import { ChevronUp, ChevronDown } from "lucide-react";

export const NumberSpinner = ({ 
  value, 
  onChange, 
  min = 0, 
  max = 100, 
  step = 1 
}) => {
  // Local state allows typing "0.5" without React snapping it back to "0" instantly
  const [localInputValue, setLocalInputValue] = useState(null);

  // Calculate precision based on step (e.g. step 0.1 -> 1 decimal place)
  const precision = step.toString().split(".")[1]?.length || 0;

  const commitValue = (val) => {
    let safeValue = Number(val.toFixed(precision));
    if (safeValue > max) safeValue = max;
    if (safeValue < min) safeValue = min;
    onChange(safeValue);
  };

  const handleIncrement = () => commitValue(value + step);
  const handleDecrement = () => commitValue(value - step);

  const handleInputChange = (e) => {
    const val = e.target.value;
    setLocalInputValue(val); 

    if (val === "" || val === "-") return;
    const num = parseFloat(val);
    if (!isNaN(num)) {
      commitValue(num);
    }
  };

  const handleBlur = () => {
    setLocalInputValue(null); // Re-sync with parent prop on blur
  };

  const displayValue = localInputValue !== null ? localInputValue : String(value);

  return (
    <div className="custom-spinner-wrapper" style={{ width: "100%" }}>
      <input
        type="text"
        inputMode="numeric"
        pattern="[0-9]*"
        className="input-base spinner-input input-number-no-spin"
        style={{ width: "100%", textAlign: "left" }} // Override width for full container fill
        value={displayValue}
        onChange={handleInputChange}
        onBlur={handleBlur}
      />
      <div className="spinner-btns">
        <button type="button" className="btn-spin" onClick={handleIncrement}>
          <ChevronUp size={12} />
        </button>
        <button type="button" className="btn-spin" onClick={handleDecrement}>
          <ChevronDown size={12} />
        </button>
      </div>
    </div>
  );
};

import React, { useState, useRef } from "react";

// Graph Constants
const PADDING = 40;
const GRAPH_MAX_X = 120;
const GRAPH_MAX_Y = 5.0;
const WIDTH = 500;
const HEIGHT = 300;

import "./PointerCurveGraph.css";

export const PointerCurveGraph = ({ data, limits, onChange, title }) => {
	const svgRef = useRef(null);
	const [dragging, setDragging] = useState(null);

	const { start, stop, scale_slow, scale_fast } = data;

	// --- COORDINATE TRANSFORMERS ---
	const valToX = (val) =>
		PADDING + (val / GRAPH_MAX_X) * (WIDTH - PADDING * 2);
	const valToY = (val) =>
		HEIGHT - PADDING - (val / GRAPH_MAX_Y) * (HEIGHT - PADDING * 2);

	const xToVal = (x) =>
		((x - PADDING) / (WIDTH - PADDING * 2)) * GRAPH_MAX_X;
	const yToVal = (y) =>
		((HEIGHT - PADDING - y) / (HEIGHT - PADDING * 2)) * GRAPH_MAX_Y;

	// Calculate SVG Points
	const p1 = { x: valToX(start), y: valToY(scale_slow) };
	const p2 = { x: valToX(stop), y: valToY(scale_fast) };

	// --- DRAG LOGIC ---
	const updateData = (modifications) => {
		const next = { ...data, ...modifications };

		// 1. Get Limits
		const lim = {
			start: {
				min: limits?.start?.min ?? 0,
				max: limits?.start?.max ?? 100,
			},
			stop: {
				min: limits?.stop?.min ?? 0,
				max: limits?.stop?.max ?? 100,
			},
			slow: {
				min: limits?.scale_slow?.min ?? 0.1,
				max: limits?.scale_slow?.max ?? 5.0,
			},
			fast: {
				min: limits?.scale_fast?.min ?? 0.1,
				max: limits?.scale_fast?.max ?? 5.0,
			},
		};

		const clamp = (v, min, max) => Math.max(min, Math.min(max, v));

		// 2. Hard Limits
		next.start = clamp(next.start, lim.start.min, lim.start.max);
		next.stop = clamp(next.stop, lim.stop.min, lim.stop.max);
		next.scale_slow = Number(
			clamp(next.scale_slow, lim.slow.min, lim.slow.max).toFixed(2)
		);
		next.scale_fast = Number(
			clamp(next.scale_fast, lim.fast.min, lim.fast.max).toFixed(2)
		);

		onChange(next);
	};

	const handlePointerMove = (e) => {
		if (!dragging) return;
		const rect = svgRef.current.getBoundingClientRect();
		// Scaling factor in case SVG is resized via CSS
		const scaleX = WIDTH / rect.width;
		const scaleY = HEIGHT / rect.height;
		
		const svgX = (e.clientX - rect.left) * scaleX;
		const svgY = (e.clientY - rect.top) * scaleY;

		let rawX = Math.round(xToVal(svgX));
		let rawY = Number(yToVal(svgY).toFixed(2));

		if (dragging === "p1") {
			if (rawX > stop) rawX = stop;
			if (rawY > scale_fast) rawY = scale_fast;
			updateData({ start: rawX, scale_slow: rawY });
		} else if (dragging === "p2") {
			if (rawX < start) rawX = start;
			if (rawY < scale_slow) rawY = scale_slow;
			updateData({ stop: rawX, scale_fast: rawY });
		}
	};

	const handlePointerDown = (point) => (e) => {
		e.preventDefault();
		setDragging(point);
		e.target.setPointerCapture(e.pointerId);
	};

	const handlePointerUp = (e) => {
		setDragging(null);
		e.target.releasePointerCapture(e.pointerId);
	};

	const pathD = `
        M ${valToX(0)} ${p1.y} 
        L ${p1.x} ${p1.y} 
        L ${p2.x} ${p2.y} 
        L ${valToX(GRAPH_MAX_X)} ${p2.y}
    `;

	return (
		<div className="graph-container">
			<svg
				ref={svgRef}
				viewBox={`0 0 ${WIDTH} ${HEIGHT}`}
				style={{ width: "100%", height: "100%", overflow: "visible", touchAction: "none" }}
				onPointerMove={handlePointerMove}
				onPointerUp={handlePointerUp}
			>
				{/* INTERNAL TITLE - CENTERED */}
				{title && (
					<text 
                        x={WIDTH / 2} 
                        y={20} 
                        className="graph-internal-title" 
                        textAnchor="middle"
                    >
						{title}
					</text>
				)}

				{/* Grid */}
				<line x1={PADDING} y1={HEIGHT - PADDING} x2={WIDTH - PADDING} y2={HEIGHT - PADDING} className="graph-grid-line" />
				<line x1={PADDING} y1={PADDING} x2={PADDING} y2={HEIGHT - PADDING} className="graph-grid-line" />

				{/* Axis Titles */}
				<text x={WIDTH - PADDING} y={HEIGHT - 10} className="graph-axis-title" textAnchor="end">
					Input Speed (deg/s)
				</text>
                {/* Moved Y-axis title down slightly (y=32) to clear the main title area */}
				<text x={10} y={32} className="graph-axis-title" textAnchor="start">
					Multiplier (x)
				</text>

				{/* Curve */}
				<path d={pathD} className="graph-curve" />
				<line x1={p1.x} y1={p1.y} x2={p1.x} y2={HEIGHT - PADDING} className="graph-drop-line" />
				<line x1={p2.x} y1={p2.y} x2={p2.x} y2={HEIGHT - PADDING} className="graph-drop-line" />

				{/* Handles */}
				<circle cx={p1.x} cy={p1.y} r="8" className="graph-handle" onPointerDown={handlePointerDown("p1")} />
				<circle cx={p2.x} cy={p2.y} r="8" className="graph-handle" onPointerDown={handlePointerDown("p2")} />

				{/* Visual Labels */}
				<text x={p1.x} y={HEIGHT - PADDING + 20} className="graph-label-text" textAnchor="middle">{start}</text>
				<text x={p2.x} y={HEIGHT - PADDING + 20} className="graph-label-text" textAnchor="middle">{stop}</text>
				<text x={PADDING - 10} y={p1.y + 4} className="graph-label-text" textAnchor="end">{scale_slow}</text>
				<text x={PADDING - 10} y={p2.y + 4} className="graph-label-text" textAnchor="end">{scale_fast}</text>
			</svg>
		</div>
	);
};

import React, { useState } from "react";
import {
	ChevronRight,
	ChevronDown,
	Box,
	Folder,
	Plus,
	Trash2,
	TrendingUp,
	Info, // <--- NEW IMPORT
} from "lucide-react";
import LeafNode from "./LeafNode";
import { useConfig } from "../../context/ConfigContext";
import { getSchema } from "../../utils/helpers";
import { SensitivityModal } from "../modals/SensitivityModal";
import "./Tree.css";

const JsonNodeRenderer = ({ name, data, path, isRootItem }) => {
	const { isAdvanced, addConfigItem, deleteConfigItem, setConfigValue } =
		useConfig();

	const [isCollapsed, setIsCollapsed] = useState(
		!isRootItem && path.length > (isRootItem ? 0 : 2)
	);

	const [showGraph, setShowGraph] = useState(false);

	const currentSchema = getSchema(path);
	const isGroup = typeof data === "object" && data !== null;
	const isArray = Array.isArray(data);
	const label = currentSchema?.label || name;

	// <--- NEW: Extract description from schema
	const description = currentSchema?.description;

	const isPointerNode = path[path.length - 1] === "pointer";

	const handleDashboardUpdate = (newPointerData) => {
		setConfigValue(path, newPointerData);
	};

	if (
		currentSchema?.mode === "hidden" ||
		(currentSchema?.mode === "advanced" && !isAdvanced)
	) {
		return null;
	}

	if (!isGroup && !Array.isArray(data)) {
		return (
			<LeafNode
				name={name}
				value={data}
				path={path}
				schema={currentSchema}
			/>
		);
	}

	const entries = isArray
		? data
				.map((item, index) => {
					if (currentSchema?.hiddenIndices?.includes(index)) return null;
					const customName = currentSchema?.names?.[index];
					const displayName = customName || `${index}`;
					return [`${index}`, item, displayName];
				})
				.filter(Boolean)
		: Object.entries(data).map(([key, val]) => [key, val, key]);

	const handleAdd = (e) => {
		e.stopPropagation();
		addConfigItem(path);
		setIsCollapsed(false);
	};

	return (
		<div className="node-group">
			{!isRootItem && (
				<div
					className="group-header"
					onClick={() => setIsCollapsed(!isCollapsed)}
				>
					{isCollapsed ? <ChevronRight size={16} /> : <ChevronDown size={16} />}
					{isArray ? (
						<Box
							size={16}
							className="text-amber-500"
						/>
					) : (
						<Folder
							size={16}
							className="text-indigo-500"
						/>
					)}
					<span
						className="group-label label-text"
						style={{ fontSize: "0.95rem", fontWeight: 600 }}
					>
						{label}
					</span>

					{/* <--- NEW: RENDER FOLDER TOOLTIP ---> */}
					{description && (
						<div
							className="tooltip-wrapper"
							onClick={(e) => e.stopPropagation()} // Prevent collapse when clicking tooltip
							style={{ marginLeft: "6px" }}
						>
							<Info size={14} />
							<div className="tooltip-content">{description}</div>
						</div>
					)}

					{/* --- DASHBOARD BUTTON --- */}
					{isPointerNode && (
						<button
							className="btn-ghost"
							style={{ marginLeft: "0.5rem", color: "var(--primary)" }}
							onClick={(e) => {
								e.stopPropagation();
								setShowGraph(true);
							}}
							// CHANGE: Replaced title="..." with data-tooltip="..."
							data-tooltip="Open Pointer Dashboard"
						>
							<TrendingUp size={16} />
						</button>
					)}

					<span
						className="group-meta"
						style={{
							fontSize: "0.75rem",
							color: "var(--text-sub)",
							marginLeft: "auto",
						}}
					>
						{entries.length} items
					</span>

					<div
						style={{ marginLeft: "1rem", display: "flex", gap: "0.5rem" }}
						onClick={(e) => e.stopPropagation()}
					>
						{isArray && currentSchema?.allowAdd && (
							<button
								className="btn-ghost"
								onClick={handleAdd}
								title="Add Item"
							>
								<Plus size={16} />
							</button>
						)}
					</div>
				</div>
			)}

			{(!isCollapsed || isRootItem) && (
				<div
					className="group-children"
					style={
						isRootItem
							? { marginLeft: 0, borderLeft: "none", paddingLeft: 0 }
							: undefined
					}
				>
					{entries.map(([key, value, displayName]) => (
						<div
							key={key}
							style={{ position: "relative" }}
						>
							<JsonNodeRenderer
								name={displayName}
								data={value}
								path={[...path, key]}
								isRootItem={false}
							/>
							{isArray && currentSchema?._item?.allowDelete && (
								<button
									className="btn-ghost"
									style={{ position: "absolute", right: "10px", top: "10px" }}
									onClick={() => deleteConfigItem([...path, key])}
								>
									<Trash2
										size={16}
										color="var(--danger)"
									/>
								</button>
							)}
						</div>
					))}
				</div>
			)}

			{isPointerNode && (
				<SensitivityModal
					isOpen={showGraph}
					onClose={() => setShowGraph(false)}
					data={data}
					schema={currentSchema}
					onSave={handleDashboardUpdate}
				/>
			)}
		</div>
	);
};

export default JsonNodeRenderer;

import React, { useState, useEffect } from "react";
// Import Icons
import { FileText, Info, ChevronUp, ChevronDown } from "lucide-react";
import { useConfig } from "../../context/ConfigContext";
import "./Tree.css";
const LeafNode = ({ name, value, path, schema }) => {
	const { setConfigValue, isAdvanced, actionList } = useConfig();

	// Local state for the number input to allow typing decimals (e.g. "0.5")
	// without the parent state snapping it back to "0" instantly.
	const [localInputValue, setLocalInputValue] = useState(null);

	// 1. Visibility Check
	if (
		schema?.mode === "hidden" ||
		(schema?.mode === "advanced" && !isAdvanced)
	) {
		return null;
	}

	const type = typeof value;
	const isReadOnly = schema?.readOnly;
	const isActionBinding = schema?.type === "action_binding";
	const hasOptions = schema?.options && Array.isArray(schema.options);

	// Label Logic
	let label = name;
	if (schema?.label) {
		label = isActionBinding ? `${schema.label} ${name}` : schema.label;
	}

	const handleChange = (newVal) => setConfigValue(path, newVal);

	const renderControl = () => {
		// A. Read Only
		// FIX: We explicitly add readOnly={true} to silence the console error
		if (isReadOnly) {
			return (
				<input
					className="input-base"
					disabled
					readOnly={true}
					value={String(value)}
				/>
			);
		}

		// B. Action Binding
		if (isActionBinding) {
			return (
				<select
					className="input-base"
					style={{ cursor: "pointer" }}
					value={value}
					onChange={(e) => handleChange(parseInt(e.target.value, 10))}
				>
					{actionList &&
						actionList.map((action, idx) => (
							<option
								key={idx}
								value={idx}
							>
								{idx}: {action.label || action.command}
							</option>
						))}
				</select>
			);
		}

		// C. Custom Options
		if (hasOptions) {
			return (
				<select
					className="input-base"
					style={{ cursor: "pointer" }}
					value={value}
					onChange={(e) => handleChange(e.target.value)}
				>
					{schema.options.map((opt) => (
						<option
							key={opt.value}
							value={opt.value}
						>
							{opt.label} ({opt.value})
						</option>
					))}
				</select>
			);
		}

		// D. Boolean
		if (type === "boolean") {
			return (
				<label className="toggle-switch">
					<input
						className="toggle-input"
						type="checkbox"
						checked={value}
						onChange={(e) => handleChange(e.target.checked)}
					/>
					<div className="toggle-track">
						<div className="toggle-thumb" />
					</div>
					<span
						style={{
							fontSize: "0.85rem",
							color: value ? "var(--primary)" : "var(--text-sub)",
						}}
					>
						{value ? "ON" : "OFF"}
					</span>
				</label>
			);
		}

		// E. Number (Slider + Custom Spinner)
		if (type === "number") {
			const min = schema?.min ?? 0;
			const max = schema?.max ?? 100;
			const step = schema?.step ?? (Number.isInteger(value) ? 1 : 0.1);

			// Logic to prevent JS floating point errors
			const precision = step.toString().split(".")[1]?.length || 0;

			// Helper to commit change to global config
			const commitValue = (val) => {
				let safeValue = Number(val.toFixed(precision));
				if (safeValue > max) safeValue = max;
				if (safeValue < min) safeValue = min;
				handleChange(safeValue);
			};

			const handleIncrement = () => commitValue(value + step);
			const handleDecrement = () => commitValue(value - step);

			// Handle manual typing
			const handleInputChange = (e) => {
				const val = e.target.value;
				setLocalInputValue(val); // Update display immediately so user can type

				// Only commit to global config if it's a valid completed number
				if (val === "" || val === "-") return;
				const num = parseFloat(val);
				if (!isNaN(num)) {
					commitValue(num);
				}
			};

			// On blur, clear local override so it re-syncs with source of truth
			const handleBlur = () => {
				setLocalInputValue(null);
			};

			// Determine what to show: local typing state OR global source of truth
			const displayValue =
				localInputValue !== null ? localInputValue : String(value);

			return (
				<div className="range-group">
					<input
						type="range"
						className="range-slider"
						min={min}
						max={max}
						step={step}
						value={value}
						onChange={(e) => handleChange(parseFloat(e.target.value))}
					/>
					<div className="custom-spinner-wrapper">
						<input
							type="text"
							inputMode="numeric"
							pattern="[0-9]*"
							className="input-base spinner-input input-number-no-spin"
							value={displayValue}
							onChange={handleInputChange}
							onBlur={handleBlur}
						/>
						<div className="spinner-btns">
							<button
								type="button"
								className="btn-spin"
								onClick={handleIncrement}
							>
								<ChevronUp size={12} />
							</button>
							<button
								type="button"
								className="btn-spin"
								onClick={handleDecrement}
							>
								<ChevronDown size={12} />
							</button>
						</div>
					</div>
				</div>
			);
		}

		// F. Default Text
		return (
			<input
				type="text"
				className="input-base"
				value={value}
				onChange={(e) => handleChange(e.target.value)}
			/>
		);
	};

	return (
		<div className="node-leaf">
			<div className="label-cell">
				<FileText
					size={16}
					style={{ opacity: 0.5, flexShrink: 0 }}
				/>
				<span title={label} className="label-text">{label}</span>
				{isReadOnly && <span className="badge-readonly">LOCK</span>}
				{schema?.description && (
					<div className="tooltip-wrapper">
						<Info size={14} />
						<div className="tooltip-content">{schema.description}</div>
					</div>
				)}
			</div>
			<div
				className="input-cell"
				style={{ width: "100%", minWidth: 0 }}
			>
				{renderControl()}
			</div>
			<div></div>
		</div>
	);
};

export default LeafNode;

import React, { createContext, useContext, useState, useEffect } from "react";
import { CONFIG_DATA } from "../constants/schemas";

const ConfigContext = createContext();

export const useConfig = () => {
	const context = useContext(ConfigContext);
	if (!context) {
		throw new Error("useConfig must be used within a ConfigProvider");
	}
	return context;
};

export const ConfigProvider = ({ children }) => {
	// --- STATE ---
	const [data, setData] = useState(CONFIG_DATA);
	const [isAdvanced, setIsAdvanced] = useState(false);
	const [isDarkMode, setIsDarkMode] = useState(false);

	// --- EFFECTS ---
	useEffect(() => {
		if (isDarkMode) document.body.setAttribute("data-theme", "dark");
		else document.body.removeAttribute("data-theme");
	}, [isDarkMode]);

	// --- HELPERS ---
	const updateData = (cb) => {
		setData((prev) => {
			const next = JSON.parse(JSON.stringify(prev));
			cb(next);
			return next;
		});
	};

	// --- ACTIONS ---
	const setConfigValue = (path, val) => {
		updateData((root) => {
			let curr = root;
			for (let i = 0; i < path.length - 1; i++) curr = curr[path[i]];
			curr[path[path.length - 1]] = val;
		});
	};

	const addConfigItem = (path) => {
		updateData((root) => {
			let curr = root;
			for (let i = 0; i < path.length; i++) curr = curr[path[i]];
			if (Array.isArray(curr)) {
				const template = curr.length ? JSON.parse(JSON.stringify(curr[0])) : {};
				if (template.name) template.name += " (Copy)";
				curr.push(template);
			}
		});
	};

	const deleteConfigItem = (path) => {
		updateData((root) => {
			let curr = root;
			for (let i = 0; i < path.length - 1; i++) curr = curr[path[i]];
			const key = path[path.length - 1];
			if (Array.isArray(curr)) curr.splice(parseInt(key), 1);
		});
	};

	const updateActions = (newActions) => {
		updateData((root) => {
			root.global_info.action_list = newActions;
		});
	};

	const resetConfig = () => setData(CONFIG_DATA);

	// --- EXPORT ---
	const value = {
		data,
		isAdvanced,
		isDarkMode,
		actionList: data.global_info?.action_list || [],
		setIsAdvanced,
		setIsDarkMode,
		setConfigValue,
		addConfigItem,
		deleteConfigItem,
		updateActions,
		resetConfig,
	};

	return (
		<ConfigContext.Provider value={value}>{children}</ConfigContext.Provider>
	);
};

import React from "react";
import { Coffee, MousePointer, Layers } from "lucide-react";
import { MOUSE_OPTS, KEY_TO_HID, MODIFIERS } from "../constants/mappings";
import { UI_SCHEMA, ACTION_SCHEMA } from "../constants/schemas";

export const getSchema = (path) => {
	let current = UI_SCHEMA;
	for (let i = 0; i < path.length; i++) {
		const key = path[i];
		if (!current) return null;
		if (current[key]) {current = current[key]; continue;}
		else if (current._item && !isNaN(parseInt(key))) current = current._item;
		else return null;
	}
	return current;
};

export const getHidLabel = (code) => {
	const mouse = MOUSE_OPTS.find((m) => m.value === code);
	if (mouse) return mouse.label.toUpperCase();
	const key = Object.keys(KEY_TO_HID).find((k) => KEY_TO_HID[k] === code);
	if (key) return key.replace(/[{}]/g, "").toUpperCase();
	return `HID ${code}`;
};

export const getModifierLabel = (mask) => {
	if (mask === 0) return "NONE";
	return MODIFIERS.filter((m) => (mask & m.mask) !== 0)
		.map((m) => m.label)
		.join(" + ");
};

export const getActionSummary = (action) => {
	if (!action) return null;
	if (action.command === "button") {
		const [actor, type, nested] = action.args || [0, "tap", [0, 0]];
		const [id, mask] = Array.isArray(nested) ? nested : [0, 0];
		const name = getHidLabel(id);
		const mods = getModifierLabel(mask);
		return (
			<div style={{ display: "flex", gap: "5px", flexWrap: "wrap" }}>
				<span className="arg-pill">{actor === 0 ? "MOUSE" : "KBD"}</span>
				<span className="arg-pill">{type}</span>
				{mods !== "NONE" && (
					<span
						className="arg-pill"
						style={{ color: "var(--primary)" }}
					>
						{mods} +
					</span>
				)}
				<span
					className="arg-pill"
					style={{ fontWeight: 700 }}
				>
					{name}
				</span>
			</div>
		);
	}
	if (action.command === "scroll") {
		const type = action.args ? action.args[0] : "fixed";
		return <span className="arg-pill">{type} mode</span>;
	}
	return <span className="arg-pill">Simple Action</span>;
};
